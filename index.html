// ================= GRIPPER (0° - 90° WEBPAGE SLIDER) ================= void setGripAngle(int targetAngle) { targetAngle = constrain(targetAngle, 0, 90); if (targetAngle > currentGripAngle) { for (int angle = currentGripAngle; angle <= targetAngle; angle++) { int servoAngle = map(angle, 0, 90, GRIP_SERVO_OPEN, GRIP_SERVO_CLOSED); gripServo.write(servoAngle); delay(15); } } else if (targetAngle < currentGripAngle) { for (int angle = currentGripAngle; angle >= targetAngle; angle--) { int servoAngle = map(angle, 0, 90, GRIP_SERVO_OPEN, GRIP_SERVO_CLOSED); gripServo.write(servoAngle); delay(15); } } else { int servoAngle = map(targetAngle, 0, 90, GRIP_SERVO_OPEN, GRIP_SERVO_CLOSED); gripServo.write(servoAngle); } currentGripAngle = targetAngle; }
